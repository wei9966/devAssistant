<script>
    const { createApp, ref, computed, onMounted, onUnmounted, nextTick } = Vue;

    createApp({
        setup() {
            const isVisible = ref(false);
            const searchQuery = ref('');
            const selectedIndex = ref(0);
            const activeTag = ref('All');
            const searchInput = ref(null);

            // 模拟数据
            const tags = [
                { name: 'All', icon: 'ph-squares-four' },
                { name: 'Dev Tools', icon: 'ph-code' },
                { name: 'AI', icon: 'ph-magic-wand' },
                { name: 'Database', icon: 'ph-database' },
                { name: 'Office', icon: 'ph-briefcase' }
            ];

            const allApps = [
                { id: 1, name: 'VS Code', icon: 'ph-code', category: 'Dev Tools', colorClass: 'bg-blue-600' },
                { id: 2, name: 'Terminal', icon: 'ph-terminal-window', category: 'Dev Tools', colorClass: 'bg-slate-700' },
                { id: 3, name: 'Chrome', icon: 'ph-globe', category: 'Dev Tools', colorClass: 'bg-yellow-600' },
                { id: 4, name: 'SQL Studio', icon: 'ph-database', category: 'Database', colorClass: 'bg-red-600' },
                { id: 5, name: 'ChatGPT', icon: 'ph-robot', category: 'AI', colorClass: 'bg-emerald-600' },
                { id: 6, name: 'Docker', icon: 'ph-cube', category: 'Dev Tools', colorClass: 'bg-blue-500' },
                { id: 7, name: 'Postman', icon: 'ph-paper-plane-tilt', category: 'Dev Tools', colorClass: 'bg-orange-600' },
                { id: 8, name: 'Notion', icon: 'ph-notebook', category: 'Office', colorClass: 'bg-black border border-white/20' },
                { id: 9, name: 'Figma', icon: 'ph-paint-brush', category: 'Office', colorClass: 'bg-purple-600' },
                { id: 10, name: 'Redis', icon: 'ph-hard-drives', category: 'Database', colorClass: 'bg-red-500' },
                { id: 11, name: 'Spotify', icon: 'ph-spotify-logo', category: 'Office', colorClass: 'bg-green-500' },
                { id: 12, name: 'Calculator', icon: 'ph-calculator', category: 'Office', colorClass: 'bg-slate-600' },
                { id: 13, name: 'Github', icon: 'ph-github-logo', category: 'Dev Tools', colorClass: 'bg-slate-800' },
                { id: 14, name: 'Settings', icon: 'ph-gear', category: 'Office', colorClass: 'bg-slate-500' },
                { id: 15, name: 'Image Gen', icon: 'ph-image', category: 'AI', colorClass: 'bg-pink-600' },
            ];

            // 过滤逻辑
            const filteredApps = computed(() => {
                let result = allApps;
                
                // Tag 过滤
                if (activeTag.value !== 'All') {
                    result = result.filter(app => app.category === activeTag.value);
                }

                // 搜索 过滤
                if (searchQuery.value) {
                    const q = searchQuery.value.toLowerCase();
                    result = result.filter(app => app.name.toLowerCase().includes(q));
                }

                return result;
            });

            // 方法
            const toggleLauncher = () => {
                isVisible.value = !isVisible.value;
                if (isVisible.value) {
                    searchQuery.value = '';
                    selectedIndex.value = 0;
                    // 必须等待 DOM 渲染后聚焦
                    nextTick(() => {
                        searchInput.value?.focus();
                    });
                }
            };

            const navigate = (direction) => {
                if (direction === 'down') {
                    if (selectedIndex.value < filteredApps.value.length - 1) {
                        selectedIndex.value++;
                    }
                } else if (direction === 'up') {
                    if (selectedIndex.value > 0) {
                        selectedIndex.value--;
                    }
                }
                // 这里可以加滚动条跟随逻辑
            };

            const openApp = (index) => {
                const app = filteredApps.value[index];
                if (app) {
                    alert(`Opening ${app.name}... (Simulation)`);
                    toggleLauncher();
                }
            };

            // 键盘事件监听
            const handleKeydown = (e) => {
                // 模拟 Alt + S (因为 Alt+Space 可能被浏览器阻止)
                if (e.altKey && (e.key === 's' || e.key === 'S')) {
                    e.preventDefault();
                    toggleLauncher();
                }
                
                // ESC 关闭
                if (e.key === 'Escape' && isVisible.value) {
                    toggleLauncher();
                }
            };

            onMounted(() => {
                window.addEventListener('keydown', handleKeydown);
            });

            onUnmounted(() => {
                window.removeEventListener('keydown', handleKeydown);
            });

            return {
                isVisible,
                searchQuery,
                selectedIndex,
                activeTag,
                tags,
                filteredApps,
                searchInput,
                toggleLauncher,
                navigate,
                openApp
            };
        }
    }).mount('#app');
</script>
