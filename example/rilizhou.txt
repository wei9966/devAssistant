import React, { useState, useMemo } from 'react';
import { 
  ChevronLeft, 
  ChevronRight, 
  Search, 
  Calendar as CalendarIcon, 
  List, 
  Columns, // Icon for Week view
  CheckCircle2, 
  Circle, 
  Clock,
  Plus,
  MoreHorizontal
} from 'lucide-react';

// --- Mock Data ---
// 增加一些数据以便展示周视图的效果
const MOCK_TASKS = [
  { id: 1, title: '智院排班数据整理', date: '2023-12-15', status: 'pending', priority: 'high', tag: '开发' },
  { id: 2, title: '江桥用户整合需求分析', date: '2023-12-15', status: 'completed', priority: 'medium', tag: '需求' },
  { id: 3, title: '松江用户整合测试', date: '2023-12-15', status: 'pending', priority: 'medium', tag: '测试' },
  { id: 4, title: '修复登录页样式Bug', date: '2023-12-14', status: 'completed', priority: 'high', tag: 'Bug' },
  { id: 5, title: '周报撰写', date: '2023-12-14', status: 'completed', priority: 'low', tag: '文档' },
  { id: 6, title: '服务器例行维护', date: '2023-12-18', status: 'pending', priority: 'high', tag: '运维' },
  { id: 7, title: '前端架构研讨会', date: '2023-12-18', status: 'pending', priority: 'medium', tag: '会议' },
  { id: 8, title: 'Q4 季度总结', date: '2023-12-25', status: 'pending', priority: 'high', tag: '文档' },
  { id: 9, title: 'UI组件库更新', date: '2023-12-05', status: 'completed', priority: 'low', tag: '开发' },
  { id: 10, title: 'Code Review', date: '2023-12-13', status: 'completed', priority: 'medium', tag: '开发' },
  { id: 11, title: '产品规划会议', date: '2023-12-13', status: 'pending', priority: 'high', tag: '会议' },
  { id: 12, title: '更新API文档', date: '2023-12-16', status: 'pending', priority: 'low', tag: '文档' },
];

const WEEKDAYS = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];

const TaskCalendar = () => {
  // State
  const [currentDate, setCurrentDate] = useState(new Date(2023, 11, 15)); // Dec 15 2023
  const [selectedDate, setSelectedDate] = useState(new Date(2023, 11, 15));
  const [searchQuery, setSearchQuery] = useState('');
  const [viewMode, setViewMode] = useState('week'); // 'month' | 'week' | 'list'

  // Helpers
  const getDaysInMonth = (year, month) => new Date(year, month + 1, 0).getDate();
  const getFirstDayOfMonth = (year, month) => new Date(year, month, 1).getDay();

  const formatDateKey = (date) => {
    return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}`;
  };

  const isSameDay = (d1, d2) => {
    return d1.getFullYear() === d2.getFullYear() &&
           d1.getMonth() === d2.getMonth() &&
           d1.getDate() === d2.getDate();
  };

  const isToday = (date) => isSameDay(date, new Date()); // Assuming system date is Dec 15 2023 for demo visualization

  // Navigation Logic
  const handlePrev = () => {
    if (viewMode === 'month') {
      setCurrentDate(new Date(currentDate.getFullYear(), currentDate.getMonth() - 1, 1));
    } else if (viewMode === 'week') {
      const newDate = new Date(currentDate);
      newDate.setDate(currentDate.getDate() - 7);
      setCurrentDate(newDate);
    }
  };

  const handleNext = () => {
    if (viewMode === 'month') {
      setCurrentDate(new Date(currentDate.getFullYear(), currentDate.getMonth() + 1, 1));
    } else if (viewMode === 'week') {
      const newDate = new Date(currentDate);
      newDate.setDate(currentDate.getDate() + 7);
      setCurrentDate(newDate);
    }
  };

  const goToToday = () => {
    const today = new Date(); // In real app, this is new Date()
    setCurrentDate(today);
    setSelectedDate(today);
  };

  // Logic for Week View Dates
  const getWeekDates = () => {
    const dates = [];
    const curr = new Date(currentDate); 
    // Adjust to start on Sunday (or Monday depending on preference, here Sunday for consistency with WeekDays array)
    const day = curr.getDay();
    const diff = curr.getDate() - day; 
    
    const startOfWeek = new Date(curr.setDate(diff));

    for (let i = 0; i < 7; i++) {
      const d = new Date(startOfWeek);
      d.setDate(startOfWeek.getDate() + i);
      dates.push(d);
    }
    return dates;
  };

  // Data Filtering
  const daysInMonth = getDaysInMonth(currentDate.getFullYear(), currentDate.getMonth());
  const firstDayOfWeek = getFirstDayOfMonth(currentDate.getFullYear(), currentDate.getMonth());

  const tasksForSelectedDate = useMemo(() => {
    const key = formatDateKey(selectedDate);
    return MOCK_TASKS.filter(t => t.date === key);
  }, [selectedDate]);

  const searchResults = useMemo(() => {
    if (!searchQuery) return [];
    return MOCK_TASKS.filter(t => t.title.toLowerCase().includes(searchQuery.toLowerCase()));
  }, [searchQuery]);

  // --- Renderers ---

  // 1. Month View
  const renderMonthView = () => {
    const days = [];
    // Empty slots
    for (let i = 0; i < firstDayOfWeek; i++) {
      days.push(<div key={`empty-${i}`} className="min-h-[6rem] bg-white/5 border border-white/5 opacity-30"></div>);
    }
    // Days
    for (let day = 1; day <= daysInMonth; day++) {
      const date = new Date(currentDate.getFullYear(), currentDate.getMonth(), day);
      const dateKey = formatDateKey(date);
      const dayTasks = MOCK_TASKS.filter(t => t.date === dateKey);
      const isSelected = isSameDay(date, selectedDate);
      const isCurrentDay = isSameDay(date, new Date());

      days.push(
        <div 
          key={day}
          onClick={() => setSelectedDate(date)}
          className={`
            min-h-[6rem] p-2 border border-white/10 relative cursor-pointer transition-all duration-200 group flex flex-col
            ${isSelected ? 'bg-indigo-500/20 border-indigo-500 shadow-[inset_0_0_20px_rgba(99,102,241,0.2)]' : 'hover:bg-white/5'}
            ${isCurrentDay ? 'bg-emerald-500/10' : ''}
          `}
        >
          <div className="flex justify-between items-start mb-1">
            <span className={`
              text-sm font-medium w-6 h-6 flex items-center justify-center rounded-full
              ${isCurrentDay ? 'bg-emerald-500 text-white' : 'text-slate-400'}
              ${isSelected && !isCurrentDay ? 'bg-indigo-500 text-white' : ''}
            `}>
              {day}
            </span>
            {dayTasks.length > 0 && (
              <span className="text-[10px] text-slate-400 bg-slate-800/80 px-1.5 py-0.5 rounded-full border border-white/10">
                {dayTasks.length}
              </span>
            )}
          </div>
          
          <div className="flex-1 flex flex-col gap-1 overflow-hidden">
            {dayTasks.slice(0, 2).map((task) => (
              <div key={task.id} className="flex items-center gap-1.5 w-full bg-white/5 rounded px-1 py-0.5">
                <div className={`w-1.5 h-1.5 rounded-full flex-shrink-0 ${task.priority === 'high' ? 'bg-rose-500' : 'bg-sky-400'}`} />
                <span className="text-[10px] text-slate-300 truncate leading-tight">
                  {task.title}
                </span>
              </div>
            ))}
            {dayTasks.length > 2 && (
              <div className="text-[10px] text-slate-500 pl-1 mt-auto">+{dayTasks.length - 2} 更多</div>
            )}
          </div>
        </div>
      );
    }
    return (
      <div className="flex-1 flex flex-col p-4 overflow-hidden">
         {/* Weekday Header */}
         <div className="grid grid-cols-7 mb-2 border-b border-white/5 pb-2">
            {WEEKDAYS.map(day => (
              <div key={day} className="text-center text-xs text-slate-500 font-medium">
                {day}
              </div>
            ))}
          </div>
          <div className="flex-1 grid grid-cols-7 auto-rows-fr gap-1 overflow-y-auto">
            {days}
          </div>
      </div>
    );
  };

  // 2. Week View (Kanban Style)
  const renderWeekView = () => {
    const dates = getWeekDates();
    
    return (
      <div className="flex-1 flex overflow-hidden p-2 gap-2">
        {dates.map((date, index) => {
          const dateKey = formatDateKey(date);
          const dayTasks = MOCK_TASKS.filter(t => t.date === dateKey);
          const isSelected = isSameDay(date, selectedDate);
          const isCurrentDay = isSameDay(date, new Date());

          return (
            <div 
              key={index}
              onClick={() => setSelectedDate(date)}
              className={`
                flex-1 min-w-[140px] flex flex-col rounded-xl transition-all duration-300 border
                ${isSelected 
                  ? 'bg-indigo-900/20 border-indigo-500/50 shadow-[0_0_30px_rgba(99,102,241,0.1)]' 
                  : 'bg-white/5 border-white/5 hover:bg-white/10'}
                ${isCurrentDay && !isSelected ? 'border-emerald-500/30 bg-emerald-500/5' : ''}
              `}
            >
              {/* Column Header */}
              <div className={`
                p-3 text-center border-b border-white/5
                ${isCurrentDay ? 'bg-gradient-to-b from-emerald-500/10 to-transparent' : ''}
              `}>
                <div className="text-xs text-slate-400 mb-1">{WEEKDAYS[date.getDay()]}</div>
                <div className={`
                  text-lg font-bold inline-flex items-center justify-center w-8 h-8 rounded-full
                  ${isCurrentDay ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/30' : 'text-slate-200'}
                  ${isSelected && !isCurrentDay ? 'bg-indigo-500 text-white' : ''}
                `}>
                  {date.getDate()}
                </div>
              </div>

              {/* Task List (Droppable area visualization) */}
              <div className="flex-1 overflow-y-auto p-2 space-y-2 custom-scrollbar">
                {dayTasks.map(task => (
                  <div key={task.id} className="bg-[#1a1d2d] p-2.5 rounded-lg border border-white/5 shadow-sm hover:border-indigo-400/50 group cursor-pointer transition-colors">
                    <div className="flex items-start justify-between gap-2 mb-1">
                      <span className={`text-xs px-1.5 py-0.5 rounded ${
                        task.priority === 'high' ? 'bg-rose-500/20 text-rose-400' : 
                        task.priority === 'medium' ? 'bg-amber-500/20 text-amber-400' :
                        'bg-slate-700 text-slate-300'
                      }`}>
                        {task.priority === 'high' ? 'P0' : task.priority === 'medium' ? 'P1' : 'P2'}
                      </span>
                      {task.status === 'completed' && <CheckCircle2 size={12} className="text-emerald-500" />}
                    </div>
                    <p className={`text-xs font-medium leading-snug ${task.status === 'completed' ? 'text-slate-500 line-through' : 'text-slate-200'}`}>
                      {task.title}
                    </p>
                  </div>
                ))}
                {dayTasks.length === 0 && (
                   <div className="h-full flex items-center justify-center opacity-0 hover:opacity-100 transition-opacity">
                      <button className="text-slate-500 hover:text-indigo-400">
                        <Plus size={20} />
                      </button>
                   </div>
                )}
              </div>
            </div>
          );
        })}
      </div>
    );
  };

  return (
    <div className="flex h-screen bg-[#0f111a] text-slate-200 font-sans overflow-hidden">
      {/* Background Ambience */}
      <div className="absolute inset-0 bg-[url('https://images.unsplash.com/photo-1534796636912-3b95b3ab5980?auto=format&fit=crop&q=80')] bg-cover opacity-20 pointer-events-none z-0" />
      <div className="absolute inset-0 bg-gradient-to-br from-[#0f111a] via-[#0f111a]/95 to-[#1a1d2d]/90 z-0" />

      {/* Main Layout Container */}
      <div className="relative z-10 flex w-full h-full p-4 gap-4">
        
        {/* Left Section: Calendar Area */}
        <div className="flex-1 flex flex-col bg-slate-900/60 backdrop-blur-md border border-white/10 rounded-2xl overflow-hidden shadow-2xl">
          
          {/* Header */}
          <header className="flex items-center justify-between p-4 border-b border-white/10">
            <div className="flex items-center gap-4">
              <h1 className="text-xl font-bold bg-clip-text text-transparent bg-gradient-to-r from-indigo-400 to-cyan-400 hidden md:block">
                任务日历
              </h1>
              
              {/* Navigation Controls */}
              <div className="flex items-center gap-1 bg-slate-800/50 rounded-lg p-1 border border-white/5 shadow-inner">
                <button onClick={handlePrev} className="p-1.5 hover:bg-white/10 rounded-md transition-colors text-slate-400 hover:text-white"><ChevronLeft size={16} /></button>
                <span className="w-32 text-center font-medium text-sm tabular-nums">
                  {currentDate.getFullYear()}年 {currentDate.getMonth() + 1}月
                  {viewMode === 'week' && <span className="text-xs text-slate-400 ml-1">(第{Math.ceil(currentDate.getDate()/7)}周)</span>}
                </span>
                <button onClick={handleNext} className="p-1.5 hover:bg-white/10 rounded-md transition-colors text-slate-400 hover:text-white"><ChevronRight size={16} /></button>
              </div>
              
              <button 
                onClick={goToToday}
                className="text-xs px-3 py-1.5 bg-indigo-600 hover:bg-indigo-500 text-white rounded-md transition-colors shadow-lg shadow-indigo-500/20 font-medium"
              >
                今天
              </button>
            </div>

            <div className="flex items-center gap-3">
              {/* Search */}
              <div className="relative group hidden sm:block">
                <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400 group-focus-within:text-indigo-400 transition-colors" />
                <input 
                  type="text" 
                  placeholder="搜索任务..." 
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  className="bg-slate-800/50 border border-white/10 rounded-full pl-9 pr-4 py-1.5 text-sm w-40 focus:w-56 focus:bg-slate-800 focus:border-indigo-500/50 outline-none transition-all duration-300"
                />
              </div>

              {/* View Switcher */}
              <div className="flex bg-slate-800/50 rounded-lg p-1 border border-white/5 shadow-inner">
                <button 
                  onClick={() => setViewMode('month')}
                  title="月视图"
                  className={`p-1.5 rounded-md transition-all duration-200 ${viewMode === 'month' ? 'bg-slate-600 text-white shadow-md' : 'text-slate-400 hover:text-white hover:bg-white/5'}`}
                >
                  <CalendarIcon size={16} />
                </button>
                <button 
                  onClick={() => setViewMode('week')}
                  title="周视图"
                  className={`p-1.5 rounded-md transition-all duration-200 ${viewMode === 'week' ? 'bg-slate-600 text-white shadow-md' : 'text-slate-400 hover:text-white hover:bg-white/5'}`}
                >
                  <Columns size={16} />
                </button>
                <button 
                  onClick={() => setViewMode('list')}
                  title="列表视图"
                  className={`p-1.5 rounded-md transition-all duration-200 ${viewMode === 'list' ? 'bg-slate-600 text-white shadow-md' : 'text-slate-400 hover:text-white hover:bg-white/5'}`}
                >
                  <List size={16} />
                </button>
              </div>
            </div>
          </header>

          {/* Main Content Area */}
          {searchQuery ? (
             <div className="flex-1 overflow-y-auto p-4 bg-slate-900/90 z-20">
               {/* Search Results reused logic */}
               <h3 className="text-sm font-medium text-slate-400 mb-3">搜索结果 ({searchResults.length})</h3>
               <div className="space-y-2">
                 {searchResults.map(task => (
                   <div key={task.id} className="flex items-center justify-between p-3 bg-white/5 rounded-lg border border-white/5 hover:border-indigo-500/30">
                     <div className="flex items-center gap-3">
                       <div className={`w-2 h-2 rounded-full ${task.priority === 'high' ? 'bg-rose-500' : 'bg-emerald-500'}`} />
                       <div>
                         <div className="text-sm text-white">{task.title}</div>
                         <div className="text-xs text-slate-500">{task.date}</div>
                       </div>
                     </div>
                     <span className="text-xs px-2 py-1 bg-white/5 rounded text-slate-400">{task.tag}</span>
                   </div>
                 ))}
               </div>
             </div>
          ) : (
            <>
              {viewMode === 'month' && renderMonthView()}
              {viewMode === 'week' && renderWeekView()}
              {viewMode === 'list' && (
                <div className="flex-1 overflow-y-auto p-4">
                  <div className="space-y-2">
                      {[...MOCK_TASKS].sort((a,b) => b.date.localeCompare(a.date)).map(task => (
                        <div key={task.id} className="flex items-center p-3 bg-white/5 rounded-lg border border-white/5 hover:bg-white/10 transition-colors">
                            <span className="text-sm w-32 text-slate-400 font-mono">{task.date}</span>
                            <span className="flex-1 text-sm font-medium text-slate-200">{task.title}</span>
                            <span className={`text-xs px-2 py-1 rounded-full ${
                              task.status === 'completed' ? 'bg-emerald-500/10 text-emerald-400' : 'bg-amber-500/10 text-amber-400'
                            }`}>
                              {task.status === 'completed' ? '已完成' : '进行中'}
                            </span>
                        </div>
                      ))}
                  </div>
                </div>
              )}
            </>
          )}
        </div>

        {/* Right Section: Task Details Panel (Shared across views) */}
        <div className="w-80 flex flex-col bg-slate-900/80 backdrop-blur-xl border border-white/10 rounded-2xl shadow-2xl">
          <div className="p-5 border-b border-white/10 bg-white/5">
            <h2 className="text-lg font-bold text-white mb-1">
              {selectedDate.getMonth() + 1}月{selectedDate.getDate()}日
            </h2>
            <div className="text-xs text-slate-400 flex items-center gap-2">
              <Clock size={12} />
              <span>{WEEKDAYS[selectedDate.getDay()]} · {tasksForSelectedDate.length} 个任务</span>
            </div>
          </div>

          <div className="flex-1 overflow-y-auto p-3 space-y-3">
            {tasksForSelectedDate.length === 0 ? (
              <div className="flex flex-col items-center justify-center h-full text-slate-500 space-y-2">
                <div className="w-12 h-12 rounded-full bg-white/5 flex items-center justify-center">
                  <CalendarIcon size={24} className="opacity-50" />
                </div>
                <p className="text-sm">今日无任务</p>
                <button className="text-xs text-indigo-400 hover:text-indigo-300 flex items-center gap-1">
                  <Plus size={12} /> 添加新任务
                </button>
              </div>
            ) : (
              tasksForSelectedDate.map(task => (
                <div key={task.id} className="group bg-[#161927] hover:bg-[#1c2030] border border-white/5 hover:border-indigo-500/30 rounded-xl p-3 transition-all duration-200 shadow-sm relative overflow-hidden">
                  <div className={`absolute left-0 top-0 bottom-0 w-1 ${task.priority === 'high' ? 'bg-rose-500' : task.priority === 'medium' ? 'bg-amber-500' : 'bg-sky-500'}`} />
                  <div className="flex items-start gap-3 pl-2">
                    <button className={`mt-0.5 ${task.status === 'completed' ? 'text-emerald-500' : 'text-slate-500 hover:text-indigo-400'}`}>
                      {task.status === 'completed' ? <CheckCircle2 size={18} /> : <Circle size={18} />}
                    </button>
                    <div className="flex-1 min-w-0">
                      <p className={`text-sm font-medium mb-1 truncate ${task.status === 'completed' ? 'text-slate-500 line-through' : 'text-slate-200'}`}>
                        {task.title}
                      </p>
                      <div className="flex items-center gap-2">
                        <span className="text-[10px] px-1.5 py-0.5 rounded bg-white/5 text-slate-400 border border-white/5">
                           {task.tag}
                        </span>
                        {task.priority === 'high' && <span className="text-[10px] text-rose-400">🔥 紧急</span>}
                      </div>
                    </div>
                    <button className="text-slate-600 hover:text-slate-300 opacity-0 group-hover:opacity-100 transition-opacity">
                      <MoreHorizontal size={14} />
                    </button>
                  </div>
                </div>
              ))
            )}
          </div>

          <div className="p-4 border-t border-white/10">
            <button className="w-full py-2.5 bg-indigo-600 hover:bg-indigo-500 active:bg-indigo-700 text-white rounded-lg text-sm font-medium transition-colors flex items-center justify-center gap-2 shadow-lg shadow-indigo-900/20">
              <Plus size={16} />
              添加任务
            </button>
          </div>
        </div>

      </div>
    </div>
  );
};

export default TaskCalendar;
