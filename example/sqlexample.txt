    <!-- 引入 Phosphor Icons (极客风格图标) -->
    <script src="https://unpkg.com/@phosphor-icons/web"></script>
    <style>
        /* 自定义滚动条，适配深色主题 */

        }

            animation: neon-pulse 3s infinite;
        }

        /* 简单的语法高亮样式 */
        .sql-keyword { color: #c084fc; font-weight: bold; } /* Purple for SELECT, FROM */
        .sql-string { color: #4ade80; } /* Green for values */

    <script type="text/babel">
        const { useState, useEffect, useRef } = React;

        // 模拟你截图中的数据
        const MOCK_DATA = [
            { id: 1, title: "查询销售订单时间范围 (最近)", sql: "SELECT * FROM SaleOrder WHERE CreatedTime >= '2025-11-24 00:00:01';", type: "SQLServer", tag: "查询", status: "success", lastUsed: "1天前" },
            { id: 2, title: "查询销售订单 (昨日)", sql: "SELECT * FROM SaleOrder WHERE CreatedTime >= '2025-11-23 00:00:01';", type: "SQLServer", tag: "查询", status: "success", lastUsed: "2天前" },
            { id: 3, title: "查询所有销售订单 - 简单", sql: "SELECT * FROM SaleOrder;", type: "MySQL", tag: "全量", status: "success", lastUsed: "2天前" },
            { id: 4, title: "统计各地区销售总额", sql: "SELECT Region, SUM(Amount) as Total FROM SaleOrder GROUP BY Region ORDER BY Total DESC;", type: "PostgreSQL", tag: "统计", status: "success", lastUsed: "3天前" },
            { id: 5, title: "查询表1数据", sql: "SELECT * FROM Table1;", type: "SQLServer", tag: "调试", status: "success", lastUsed: "2天前" },
            { id: 6, title: "更新用户状态 (慎用)", sql: "UPDATE Users SET Status = 0 WHERE LastLogin < '2024-01-01';", type: "SQLServer", tag: "维护", status: "warning", lastUsed: "1周前" },
            { id: 7, title: "删除测试数据", sql: "DELETE FROM Logs WHERE Level = 'DEBUG';", type: "MySQL", tag: "清理", status: "danger", lastUsed: "1个月前" },
        ];

        // 简单的 SQL 语法高亮组件
        const SqlHighlighter = ({ code }) => {
            const highlight = (sql) => {
                const keywords = /\b(SELECT|FROM|WHERE|GROUP BY|ORDER BY|UPDATE|SET|DELETE|INSERT|INTO|VALUES|AND|OR|AS|SUM|COUNT|LIMIT|TOP)\b/gi;
                const strings = /'([^']*)'/g;
                
                let html = sql
                    .replace(keywords, '<span class="sql-keyword">$1</span>')
                    .replace(strings, '<span class="sql-string">\'$1\'</span>');
                return { __html: html };
            };

            return (
                <pre className="font-mono text-sm leading-relaxed whitespace-pre-wrap break-words text-slate-300" dangerouslySetInnerHTML={highlight(code)} />
            );
        };

        const App = () => {
            const [query, setQuery] = useState("");
            const [selectedIndex, setSelectedIndex] = useState(0);
            const [filteredData, setFilteredData] = useState(MOCK_DATA);
            const [showToast, setShowToast] = useState(false);
            const inputRef = useRef(null);
            const listRef = useRef(null);

            // 搜索过滤逻辑
            useEffect(() => {
                const lowerQuery = query.toLowerCase();
                const results = MOCK_DATA.filter(item => 
                    item.title.toLowerCase().includes(lowerQuery) || 
                    item.sql.toLowerCase().includes(lowerQuery)
                );
                setFilteredData(results);
                setSelectedIndex(0);
            }, [query]);

            // 键盘快捷键逻辑
            useEffect(() => {
                const handleKeyDown = (e) => {
                    if (e.key === 'ArrowDown') {
                        e.preventDefault();
                        setSelectedIndex(prev => (prev + 1) % filteredData.length);
                        // 自动滚动
                        if (listRef.current) {
                           // 简单的滚动逻辑，实际项目可以使用 scrollIntoView
                        }
                    } else if (e.key === 'ArrowUp') {
                        e.preventDefault();
                        setSelectedIndex(prev => (prev - 1 + filteredData.length) % filteredData.length);
                    } else if (e.key === 'Enter') {
                        e.preventDefault();
                        if (filteredData[selectedIndex]) {
                            handleCopy(filteredData[selectedIndex].sql);
                        }
                    } else if (e.key === 'Escape') {
                         // 在这里处理关闭模态框逻辑
                         console.log("Close Modal");
                    }
                };

                window.addEventListener('keydown', handleKeyDown);
                return () => window.removeEventListener('keydown', handleKeyDown);
            }, [filteredData, selectedIndex]);

            // 聚焦输入框
            useEffect(() => {
                if(inputRef.current) inputRef.current.focus();
            }, []);

            const handleCopy = (sql) => {
                // 模拟复制
                // navigator.clipboard.writeText(sql); 
                setShowToast(true);
                setTimeout(() => setShowToast(false), 2000);
            };

            const selectedItem = filteredData[selectedIndex];

            return (
                <div className="flex items-center justify-center w-full h-full bg-[url('https://images.unsplash.com/photo-1550751827-4bd374c3f58b?q=80&w=2070&auto=format&fit=crop')] bg-cover">
                    {/* 背景遮罩 */}
                    <div className="absolute inset-0 bg-slate-950/80 backdrop-blur-sm"></div>

                    {/* Toast 提示 */}
                    {showToast && (
                        <div className="absolute top-10 left-1/2 transform -translate-x-1/2 z-50 animate-bounce">
                            <div className="bg-cyan-500 text-slate-950 px-6 py-2 rounded-full font-bold shadow-lg flex items-center gap-2">
                                <i className="ph-bold ph-check"></i>
                                SQL 已复制到剪贴板
                            </div>
                        </div>
                    )}

                    {/* 主模态框 */}
                    <div className="relative w-[900px] h-[600px] bg-slate-900 border border-slate-700 rounded-2xl shadow-2xl flex flex-col overflow-hidden neon-border">
                        
                        {/* 1. 顶部搜索栏 */}
                        <div className="h-16 border-b border-slate-800 flex items-center px-6 shrink-0 bg-slate-900/50">
                            <i className="ph ph-magnifying-glass text-slate-400 text-2xl mr-4"></i>
                            <input 
                                ref={inputRef}
                                type="text" 
                                placeholder="搜索 SQL 历史 (输入 'select', 'order', '时间'...)" 
                                className="w-full bg-transparent border-none outline-none text-xl text-slate-200 placeholder-slate-500 h-full"
                                value={query}
                                onChange={(e) => setQuery(e.target.value)}
                            />
                            <div className="flex gap-2">
                                <span className="bg-slate-800 text-slate-400 text-xs px-2 py-1 rounded border border-slate-700 font-mono">ESC 关闭</span>
                            </div>
                        </div>

                        {/* 2. 主体内容区 (双栏布局) */}
                        <div className="flex flex-1 overflow-hidden">
                            
                            {/* 左侧：列表区 (40%) */}
                            <div className="w-2/5 border-r border-slate-800 overflow-y-auto bg-slate-900/30" ref={listRef}>
                                {filteredData.length === 0 ? (
                                    <div className="p-8 text-center text-slate-500">
                                        <i className="ph ph-ghost text-4xl mb-2"></i>
                                        <p>没有找到相关 SQL</p>
                                    </div>
                                ) : (
                                    filteredData.map((item, index) => (
                                        <div 
                                            key={item.id}
                                            onClick={() => setSelectedIndex(index)}
                                            className={`
                                                group relative px-4 py-3 cursor-pointer transition-all duration-150 border-b border-slate-800/50
                                                ${index === selectedIndex ? 'bg-slate-800/80' : 'hover:bg-slate-800/40'}
                                            `}
                                        >
                                            {/* 选中态的左侧指示条 */}
                                            {index === selectedIndex && (
                                                <div className="absolute left-0 top-0 bottom-0 w-1 bg-cyan-400 shadow-[0_0_10px_#22d3ee]"></div>
                                            )}

                                            <div className="flex justify-between items-start mb-1">
                                                <span className={`font-medium truncate pr-2 ${index === selectedIndex ? 'text-cyan-50' : 'text-slate-300'}`}>
                                                    {item.title}
                                                </span>
                                                {/* 小图标/Tag */}
                                                <div className="flex gap-1 shrink-0">
                                                    {item.status === 'success' && <i className="ph-fill ph-check-circle text-green-500"></i>}
                                                    {item.status === 'warning' && <i className="ph-fill ph-warning text-orange-400"></i>}
                                                    {item.status === 'danger' && <i className="ph-fill ph-x-circle text-red-500"></i>}
                                                </div>
                                            </div>
                                            
                                            {/* SQL 预览 (截断) */}
                                            <div className="text-xs text-slate-500 font-mono truncate opacity-60">
                                                {item.sql}
                                            </div>

                                            {/* 底部元数据 */}
                                            <div className="flex items-center gap-2 mt-2">
                                                <span className={`text-[10px] px-1.5 py-0.5 rounded border ${
                                                    item.type === 'SQLServer' ? 'border-red-900/50 text-red-400 bg-red-900/20' : 
                                                    item.type === 'MySQL' ? 'border-blue-900/50 text-blue-400 bg-blue-900/20' :
                                                    'border-slate-700 text-slate-400'
                                                }`}>
                                                    {item.type}
                                                </span>
                                                <span className="text-[10px] text-slate-600 ml-auto">{item.lastUsed}</span>
                                            </div>
                                        </div>
                                    ))
                                )}
                            </div>

                            {/* 右侧：预览区 (60%) */}
                            <div className="w-3/5 bg-[#0b1120] flex flex-col relative">
                                {selectedItem ? (
                                    <>
                                        {/* 预览头部 */}
                                        <div className="h-12 border-b border-slate-800 flex items-center justify-between px-6 bg-[#0f172a]">
                                            <span className="text-slate-400 text-xs font-mono flex items-center gap-2">
                                                <i className="ph ph-code"></i>
                                                PREVIEW
                                            </span>
                                            <div className="flex gap-2">
                                                <button onClick={() => handleCopy(selectedItem.sql)} className="text-xs flex items-center gap-1 bg-cyan-500/10 hover:bg-cyan-500/20 text-cyan-400 px-3 py-1 rounded border border-cyan-500/20 transition-colors">
                                                    <i className="ph-bold ph-copy"></i> 复制
                                                </button>
                                                <button className="text-xs flex items-center gap-1 bg-slate-800 hover:bg-slate-700 text-slate-300 px-3 py-1 rounded border border-slate-700 transition-colors">
                                                    <i className="ph-bold ph-pencil-simple"></i> 编辑
                                                </button>
                                            </div>
                                        </div>

                                        {/* 代码展示区 */}
                                        <div className="flex-1 p-6 overflow-y-auto custom-scrollbar">
                                            <div className="mb-4">
                                                <h2 className="text-lg font-semibold text-slate-100 mb-1">{selectedItem.title}</h2>
                                                <div className="flex gap-2 text-xs text-slate-500">
                                                    <span>分类: {selectedItem.tag}</span>
                                                    <span>•</span>
                                                    <span>上次执行: {selectedItem.lastUsed}</span>
                                                </div>
                                            </div>
                                            
                                            <div className="bg-[#1e293b]/50 border border-slate-700/50 rounded-lg p-4 relative group">
                                                <div className="absolute top-2 right-2 opacity-0 group-hover:opacity-100 transition-opacity">
                                                    <span className="text-[10px] text-slate-500 font-mono">SQL</span>
                                                </div>
                                                <SqlHighlighter code={selectedItem.sql} />
                                            </div>

                                            {/* AI 解析或备注区域 (可选) */}
                                            <div className="mt-6 border-t border-slate-800 pt-4">
                                                <h3 className="text-xs font-bold text-slate-500 uppercase mb-2 flex items-center gap-1">
                                                    <i className="ph-fill ph-magic-wand text-purple-400"></i>
                                                    AI 备注
                                                </h3>
                                                <p className="text-sm text-slate-400 leading-relaxed">
                                                    该查询主要用于获取订单列表。注意 <span className="text-slate-300 bg-slate-800 px-1 rounded">CreatedTime</span> 字段没有建立索引可能会导致全表扫描。建议仅在非高峰期运行。
                                                </p>
                                            </div>
                                        </div>
                                    </>
                                ) : (
                                    <div className="flex flex-col items-center justify-center h-full text-slate-600">
                                        <i className="ph ph-code text-6xl mb-4 opacity-20"></i>
                                        <p>选择左侧列表查看详情</p>
                                    </div>
                                )}
                            </div>
                        </div>

                        {/* 3. 底部状态栏 */}
                        <div className="h-10 border-t border-slate-800 bg-slate-900/80 flex items-center px-4 justify-between text-[11px] text-slate-500 select-none">
                            <div className="flex gap-4">
                                <span className="flex items-center gap-1"><kbd className="bg-slate-800 px-1 rounded text-slate-300">↵</kbd> 复制</span>
                                <span className="flex items-center gap-1"><kbd className="bg-slate-800 px-1 rounded text-slate-300">Shift + ↵</kbd> 直接运行</span>
                                <span className="flex items-center gap-1"><kbd className="bg-slate-800 px-1 rounded text-slate-300">↑↓</kbd> 导航</span>
                            </div>
                            <div>
                                {filteredData.length} 条记录
                            </div>
                        </div>
                    </div>
                </div>
            );
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
